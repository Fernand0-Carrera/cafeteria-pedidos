 26class EstadoPedido(str, Enum):
 27    """Etapas de atención de un pedido (RN-NEG-06).
 28
 29    El orden de declaración es el único orden permitido de avance:
 30    Pendiente → En preparación → Listo para retirar → Entregado.
 31    """
 32
 33    PENDIENTE = "PENDIENTE"
 34    EN_PREPARACION = "EN_PREPARACION"
 35    LISTO_PARA_RETIRAR = "LISTO_PARA_RETIRAR"
 36    ENTREGADO = "ENTREGADO"
 37
 38    @property
 39    def etiqueta(self) -> str:
 40        """Nombre legible del estado, tal como aparece en la especificación."""
 41        return _ETIQUETAS_ESTADO[self]
 42
 43    def siguiente(self) -> Optional["EstadoPedido"]:
 44        """Devuelve el estado inmediatamente posterior, o ``None`` si es el último."""
 45        orden = list(EstadoPedido)
 46        posicion = orden.index(self)
 47        return orden[posicion + 1] if posicion + 1 < len(orden) else None
 48
 49
 50_ETIQUETAS_ESTADO = {
 51    EstadoPedido.PENDIENTE: "Pendiente",
 52    EstadoPedido.EN_PREPARACION: "En preparación",
 53    EstadoPedido.LISTO_PARA_RETIRAR: "Listo para retirar",
 54    EstadoPedido.ENTREGADO: "Entregado",
 55}
