app.persistencia.repositorios

Repositorios: traducen entre filas de SQLite y entidades del dominio.

Los repositorios no aplican reglas de negocio; solo leen y escriben datos. Las transacciones las controla la capa de servicios.

 47class ProductoRepositorio:
 48    """Acceso a la tabla ``producto``."""
 49
 50    def __init__(self, conexion: sqlite3.Connection):
 51        self._con = conexion
 52
 53    def listar(self, solo_disponibles: bool = False) -> list[Producto]:
 54        """Lista productos ordenados por categoría y nombre."""
 55        sql = "SELECT * FROM producto"
 56        if solo_disponibles:
 57            sql += " WHERE disponible = 1"
 58        sql += " ORDER BY categoria, nombre"
 59        return [_a_producto(f) for f in self._con.execute(sql)]
 60
 61    def obtener_por_codigo(self, codigo: str) -> Optional[Producto]:
 62        """Devuelve el producto con el código indicado o ``None``."""
 63        fila = self._con.execute("SELECT * FROM producto WHERE codigo = ?", (codigo,)).fetchone()
 64        return _a_producto(fila) if fila else None
 65
 66    def obtener_por_codigos(self, codigos: Iterable[str]) -> dict[str, Producto]:
 67        """Devuelve un diccionario código → producto para los códigos existentes."""
 68        codigos = list(codigos)
 69        if not codigos:
 70            return {}
 71        # La lista viaja como un único parámetro JSON: la sentencia SQL es fija y no
 72        # se construye concatenando texto (evita el aviso B608 de inyección SQL).
 73        filas = self._con.execute(
 74            "SELECT * FROM producto WHERE codigo IN (SELECT value FROM json_each(?))",
 75            (json.dumps(codigos),),
 76        )
 77        return {f["codigo"]: _a_producto(f) for f in filas}
 78
 79    def crear(self, producto: Producto) -> Producto:
 80        """Inserta un producto y devuelve la entidad con su ``id``."""
 81        cursor = self._con.execute(
 82            "INSERT INTO producto (codigo, nombre, categoria, precio_centavos, disponible) "
 83            "VALUES (?, ?, ?, ?, ?)",
 84            (producto.codigo, producto.nombre, producto.categoria.value,
 85             producto.precio_centavos, int(producto.disponible)),
 86        )
 87        producto.id = cursor.lastrowid
 88        return producto
 89
 90    def actualizar(self, producto: Producto) -> None:
 91        """Actualiza nombre, categoría, precio y disponibilidad de un producto."""
 92        self._con.execute(
 93            "UPDATE producto SET nombre = ?, categoria = ?, precio_centavos = ?, disponible = ? "
 94            "WHERE codigo = ?",
 95            (producto.nombre, producto.categoria.value, producto.precio_centavos,
 96             int(producto.disponible), producto.codigo),
 97        )
 98
 99    def contar(self) -> int:
100        """Número total de productos registrados."""
101        return self._con.execute("SELECT COUNT(*) FROM producto").fetchone()[0]

104class ClienteRepositorio:
105    """Acceso a la tabla ``cliente``."""
106
107    def __init__(self, conexion: sqlite3.Connection):
108        self._con = conexion
109
110    def crear(self, cliente: Cliente) -> Cliente:
111        """Inserta el cliente y le asigna un código con formato ``CLI-00001``."""
112        cursor = self._con.execute(
113            "INSERT INTO cliente (nombre, apellido, cedula, celular, correo) VALUES (?, ?, ?, ?, ?)",
114            (cliente.nombre, cliente.apellido, cliente.cedula, cliente.celular, cliente.correo),
115        )
116        cliente.id = cursor.lastrowid
117        cliente.codigo = f"CLI-{cliente.id:05d}"
118        self._con.execute("UPDATE cliente SET codigo = ? WHERE id = ?", (cliente.codigo, cliente.id))
119        return cliente
120
121    def obtener_por_id(self, cliente_id: int) -> Optional[Cliente]:
122        """Devuelve el cliente con el ``id`` indicado o ``None``."""
123        fila = self._con.execute("SELECT * FROM cliente WHERE id = ?", (cliente_id,)).fetchone()
124        return _a_cliente(fila) if fila else None
125
126    def obtener_por_cedula(self, cedula: str) -> Optional[Cliente]:
127        """Devuelve el cliente con la cédula indicada o ``None``."""
128        fila = self._con.execute("SELECT * FROM cliente WHERE cedula = ?", (cedula,)).fetchone()
129        return _a_cliente(fila) if fila else None
130
131    def existe_correo(self, correo: str) -> bool:
132        """Indica si el correo ya está registrado."""
133        return self._con.execute("SELECT 1 FROM cliente WHERE correo = ?", (correo,)).fetchone() is not None

136class PedidoRepositorio:
137    """Acceso a las tablas ``pedido``, ``detalle_pedido`` e ``historial_estado``."""
138
139    _SELECT_PEDIDO = (
140        "SELECT p.*, c.nombre || ' ' || c.apellido AS cliente_nombre "
141        "FROM pedido p JOIN cliente c ON c.id = p.cliente_id"
142    )
143
144    def __init__(self, conexion: sqlite3.Connection):
145        self._con = conexion
146
147    def existe_codigo(self, codigo_retiro: str) -> bool:
148        """Indica si ya existe un pedido con ese código de retiro."""
149        return self._con.execute(
150            "SELECT 1 FROM pedido WHERE codigo_retiro = ?", (codigo_retiro,)
151        ).fetchone() is not None
152
153    def crear(self, pedido: Pedido, usuario: str) -> Pedido:
154        """Inserta el pedido, sus detalles y el primer registro del historial."""
155        cursor = self._con.execute(
156            "INSERT INTO pedido (codigo_retiro, cliente_id, estado, fecha_registro, total_centavos) "
157            "VALUES (?, ?, ?, ?, ?)",
158            (pedido.codigo_retiro, pedido.cliente_id, pedido.estado.value,
159             pedido.fecha_registro, pedido.total_centavos),
160        )
161        pedido.id = cursor.lastrowid
162        self._con.executemany(
163            "INSERT INTO detalle_pedido (pedido_id, producto_id, codigo_producto, nombre_producto, "
164            "cantidad, precio_unitario_centavos) VALUES (?, ?, ?, ?, ?, ?)",
165            [(pedido.id, d.producto_id, d.codigo_producto, d.nombre_producto, d.cantidad,
166              d.precio_unitario_centavos) for d in pedido.detalles],
167        )
168        self.registrar_historial(pedido.id, pedido.estado, pedido.fecha_registro, usuario)
169        return pedido
170
171    def registrar_historial(self, pedido_id: int, estado: EstadoPedido, fecha: str, usuario: str) -> None:
172        """Agrega una fila al historial de estados del pedido."""
173        self._con.execute(
174            "INSERT INTO historial_estado (pedido_id, estado, fecha, usuario) VALUES (?, ?, ?, ?)",
175            (pedido_id, estado.value, fecha, usuario),
176        )
177
178    def actualizar_estado(self, pedido_id: int, estado_actual: EstadoPedido,
179                          estado_nuevo: EstadoPedido) -> bool:
180        """Cambia el estado solo si el pedido sigue en ``estado_actual``.
181
182        Returns:
183            ``True`` si se actualizó; ``False`` si otro usuario lo cambió antes
184            (control de concurrencia optimista).
185        """
186        cursor = self._con.execute(
187            "UPDATE pedido SET estado = ? WHERE id = ? AND estado = ?",
188            (estado_nuevo.value, pedido_id, estado_actual.value),
189        )
190        return cursor.rowcount == 1
191
192    def obtener(self, pedido_id: int) -> Optional[Pedido]:
193        """Devuelve el pedido con sus detalles o ``None``."""
194        fila = self._con.execute(f"{self._SELECT_PEDIDO} WHERE p.id = ?", (pedido_id,)).fetchone()
195        return self._con_detalles(fila) if fila else None
196
197    def obtener_por_codigo(self, codigo_retiro: str) -> Optional[Pedido]:
198        """Devuelve el pedido con el código de retiro indicado o ``None``."""
199        fila = self._con.execute(
200            f"{self._SELECT_PEDIDO} WHERE p.codigo_retiro = ?", (codigo_retiro,)
201        ).fetchone()
202        return self._con_detalles(fila) if fila else None
203
204    def listar_por_cliente(self, cliente_id: int) -> list[Pedido]:
205        """Pedidos del cliente, del más reciente al más antiguo."""
206        filas = self._con.execute(
207            f"{self._SELECT_PEDIDO} WHERE p.cliente_id = ? ORDER BY p.id DESC", (cliente_id,)
208        )
209        return [self._con_detalles(f) for f in filas]
210
211    def listar(self, estado: Optional[EstadoPedido] = None) -> list[Pedido]:
212        """Pedidos filtrados por estado, del más antiguo al más reciente (orden de atención)."""
213        if estado is None:
214            filas = self._con.execute(f"{self._SELECT_PEDIDO} ORDER BY p.id")
215        else:
216            filas = self._con.execute(
217                f"{self._SELECT_PEDIDO} WHERE p.estado = ? ORDER BY p.id", (estado.value,)
218            )
219        return [self._con_detalles(f) for f in filas]
220
221    def contar_por_estado(self) -> dict[EstadoPedido, int]:
222        """Cantidad de pedidos en cada estado."""
223        conteo = {estado: 0 for estado in EstadoPedido}
224        for fila in self._con.execute("SELECT estado, COUNT(*) AS n FROM pedido GROUP BY estado"):
225            conteo[EstadoPedido(fila["estado"])] = fila["n"]
226        return conteo
227
228    def historial(self, pedido_id: int) -> list[CambioEstado]:
229        """Cambios de estado del pedido en orden cronológico."""
230        filas = self._con.execute(
231            "SELECT estado, fecha, usuario FROM historial_estado WHERE pedido_id = ? ORDER BY id",
232            (pedido_id,),
233        )
234        return [CambioEstado(EstadoPedido(f["estado"]), f["fecha"], f["usuario"]) for f in filas]
235
236    def _con_detalles(self, fila: sqlite3.Row) -> Pedido:
237        detalles = [
238            DetallePedido(
239                producto_id=d["producto_id"],
240                codigo_producto=d["codigo_producto"],
241                nombre_producto=d["nombre_producto"],
242                cantidad=d["cantidad"],
243                precio_unitario_centavos=d["precio_unitario_centavos"],
244            )
245            for d in self._con.execute(
246                "SELECT * FROM detalle_pedido WHERE pedido_id = ? ORDER BY id", (fila["id"],)
247            )
248        ]
249        return Pedido(
250            id=fila["id"],
251            codigo_retiro=fila["codigo_retiro"],
252            cliente_id=fila["cliente_id"],
253            estado=EstadoPedido(fila["estado"]),
254            fecha_registro=fila["fecha_registro"],
255            total_centavos=fila["total_centavos"],
256            detalles=detalles,
257            cliente_nombre=fila["cliente_nombre"],
258        )

140    _SELECT_PEDIDO = (
141        "SELECT p.*, c.nombre || ' ' || c.apellido AS cliente_nombre "
142        "FROM pedido p JOIN cliente c ON c.id = p.cliente_id"
143    )

146        self._con = conexion

148    def existe_codigo(self, codigo_retiro: str) -> bool:
149        """Indica si ya existe un pedido con ese código de retiro."""
150        return self._con.execute(
151            "SELECT 1 FROM pedido WHERE codigo_retiro = ?", (codigo_retiro,)
152        ).fetchone() is not None
153
154    def crear(self, pedido: Pedido, usuario: str) -> Pedido:
155        """Inserta el pedido, sus detalles y el primer registro del historial."""
156        cursor = self._con.execute(
157            "INSERT INTO pedido (codigo_retiro, cliente_id, estado, fecha_registro, total_centavos) "
158            "VALUES (?, ?, ?, ?, ?)",
159            (pedido.codigo_retiro, pedido.cliente_id, pedido.estado.value,
160             pedido.fecha_registro, pedido.total_centavos),
161        )
162        pedido.id = cursor.lastrowid
163        self._con.executemany(
164            "INSERT INTO detalle_pedido (pedido_id, producto_id, codigo_producto, nombre_producto, "
165            "cantidad, precio_unitario_centavos) VALUES (?, ?, ?, ?, ?, ?)",
166            [(pedido.id, d.producto_id, d.codigo_producto, d.nombre_producto, d.cantidad,
167              d.precio_unitario_centavos) for d in pedido.detalles],
168        )
169        self.registrar_historial(pedido.id, pedido.estado, pedido.fecha_registro, usuario)
170        return pedido
171
172    def registrar_historial(self, pedido_id: int, estado: EstadoPedido, fecha: str, usuario: str) -> None:
173        """Agrega una fila al historial de estados del pedido."""
174        self._con.execute(
175            "INSERT INTO historial_estado (pedido_id, estado, fecha, usuario) VALUES (?, ?, ?, ?)",
176            (pedido_id, estado.value, fecha, usuario),
177        )
178
179    def actualizar_estado(self, pedido_id: int, estado_actual: EstadoPedido,
180                          estado_nuevo: EstadoPedido) -> bool:
181        """Cambia el estado solo si el pedido sigue en ``estado_actual``.
182
183        Returns:
184            ``True`` si se actualizó; ``False`` si otro usuario lo cambió antes
185            (control de concurrencia optimista).
186        """
187        cursor = self._con.execute(
188            "UPDATE pedido SET estado = ? WHERE id = ? AND estado = ?",
189            (estado_nuevo.value, pedido_id, estado_actual.value),
190        )
191        return cursor.rowcount == 1
192
193    def obtener(self, pedido_id: int) -> Optional[Pedido]:
194        """Devuelve el pedido con sus detalles o ``None``."""
195        fila = self._con.execute(f"{self._SELECT_PEDIDO} WHERE p.id = ?", (pedido_id,)).fetchone()
196        return self._con_detalles(fila) if fila else None

198    def obtener_por_codigo(self, codigo_retiro: str) -> Optional[Pedido]:
199        """Devuelve el pedido con el código de retiro indicado o ``None``."""

205    def listar_por_cliente(self, cliente_id: int) -> list[Pedido]:
206        """Pedidos del cliente, del más reciente al más antiguo."""
207        filas = self._con.execute(
208            f"{self._SELECT_PEDIDO} WHERE p.cliente_id = ? ORDER BY p.id DESC", (cliente_id,)
209        )
210        return [self._con_detalles(f) for f in filas]
211
212    def listar(self, estado: Optional[EstadoPedido] = None) -> list[Pedido]:
213        """Pedidos filtrados por estado, del más antiguo al más reciente (orden de atención)."""
214        if estado is None:
215            filas = self._con.execute(f"{self._SELECT_PEDIDO} ORDER BY p.id")
216        else:
217            filas = self._con.execute(
218                f"{self._SELECT_PEDIDO} WHERE p.estado = ? ORDER BY p.id", (estado.value,)
219            )
220        return [self._con_detalles(f) for f in filas]
221
222    def contar_por_estado(self) -> dict[EstadoPedido, int]:
223        """Cantidad de pedidos en cada estado."""
224        conteo = {estado: 0 for estado in EstadoPedido}
225        for fila in self._con.execute("SELECT estado, COUNT(*) AS n FROM pedido GROUP BY estado"):
226            conteo[EstadoPedido(fila["estado"])] = fila["n"]
227        return conteo
228
229    def historial(self, pedido_id: int) -> list[CambioEstado]:
230        """Cambios de estado del pedido en orden cronológico."""
231        filas = self._con.execute(
232            "SELECT estado, fecha, usuario FROM historial_estado WHERE pedido_id = ? ORDER BY id",
233            (pedido_id,),
234        )
235        return [CambioEstado(EstadoPedido(f["estado"]), f["fecha"], f["usuario"]) for f in filas]

237    def _con_detalles(self, fila: sqlite3.Row) -> Pedido:
238        detalles = [
239            DetallePedido(
240                producto_id=d["producto_id"],
241                codigo_producto=d["codigo_producto"],
242                nombre_producto=d["nombre_producto"],
243                cantidad=d["cantidad"],
244                precio_unitario_centavos=d["precio_unitario_centavos"],
245            )
246            for d in self._con.execute(
247                "SELECT * FROM detalle_pedido WHERE pedido_id = ? ORDER BY id", (fila["id"],)
248            )
249        ]
250        return Pedido(
251            id=fila["id"],
252            codigo_retiro=fila["codigo_retiro"],
253            cliente_id=fila["cliente_id"],
254            estado=EstadoPedido(fila["estado"]),
255            fecha_registro=fila["fecha_registro"],
256            total_centavos=fila["total_centavos"],
257            detalles=detalles,
258            cliente_nombre=fila["cliente_nombre"],
259        )
