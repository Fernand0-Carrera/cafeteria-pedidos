53 def interpretar_cantidades(self, entradas: Mapping[str, str]) -> dict[str, int]: 

54 """Convierte las cantidades escritas en el formulario a enteros. 

55 

56 Los campos vacíos o con ``0`` se interpretan como «producto no seleccionado». 

57 

58 Raises: 

59 CantidadInvalidaError: Si alguna cantidad no es un número entero positivo. 

60 """ 

61 cantidades: dict[str, int] = {} 

62 for codigo, texto in entradas.items(): 

63 texto = (texto or "").strip() 

64 if texto in ("", "0"): 

65 continue 

66 if not re.fullmatch(r"[0-9]+", texto): 

67 raise CantidadInvalidaError( 

68 f"La cantidad de {codigo} debe ser un número entero positivo (recibido: «{texto}»)." 

69 ) 

70 cantidades[codigo] = int(texto) 

71 return cantidades

105 def confirmar_pedido(self, cliente_id: int, items: Mapping[str, int], 

106 total_mostrado_centavos: Optional[int] = None) -> Pedido: 

107 """Registra el pedido en estado Pendiente con un código de retiro único (RU-04). 

108 

109 Se vuelven a validar las reglas al confirmar porque el menú pudo cambiar 

110 mientras el cliente revisaba su selección. 

111 

112 Args: 

113 cliente_id: Cliente que confirma. 

114 items: Código de producto → cantidad. 

115 total_mostrado_centavos: Total que vio el cliente; si los precios cambiaron 

116 y el total es distinto, el pedido no se confirma. 

117 

118 Raises: 

119 TotalDesactualizadoError: Si el total vigente difiere del mostrado (RN-NEG-03). 

120 ErrorNegocio: Cualquier error de :meth:`calcular_resumen`. 

121 """ 

122 resumen = self.calcular_resumen(items) 

123 if total_mostrado_centavos is not None and resumen.total_centavos != total_mostrado_centavos: 

124 raise TotalDesactualizadoError( 

125 "Los precios cambiaron mientras revisaba su pedido. " 

126 f"El nuevo total es {formatear_centavos(resumen.total_centavos)}; revíselo y confirme otra vez." 

127 ) 

128 pedido = Pedido( 

129 codigo_retiro=self._codigo_unico(), 

130 cliente_id=cliente_id, 

131 estado=EstadoPedido.PENDIENTE, 

132 fecha_registro=self._reloj(), 

133 total_centavos=resumen.total_centavos, 

134 detalles=resumen.detalles, 

135 ) 

136 with self._con: 

137 self._pedidos.crear(pedido, usuario="cliente") 

138 return self._pedidos.obtener(pedido.id)
