32 ClienteDuplicadoError: Si la cédula o el correo ya están registrados.

41 if self._clientes.obtener_por_cedula(cliente.cedula): 

42 raise ClienteDuplicadoError("Ya existe un cliente registrado con esa cédula.") 

43 if self._clientes.existe_correo(cliente.correo): 

44 raise ClienteDuplicadoError("Ya existe un cliente registrado con ese correo.") 

45 with self._con:
